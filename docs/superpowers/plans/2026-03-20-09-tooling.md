### Task 4: Add save feedback

- [ ] Find current save handlers (they are likely silent on success):
```bash
grep -n "onSave\|saveAsset\|handleSave\|\.then(" auxserver/static/asseteditor.js | head -20
```

- [ ] For each save handler, add visible feedback after the fetch:
```javascript
async function saveAsset(data) {
  try {
    const res = await fetch('/api/assets', { method: 'POST', body: JSON.stringify(data), headers: {'Content-Type': 'application/json'} });
    if (!res.ok) throw new Error(`Server error: ${res.status}`);
    showFeedback('Saved!', 'success');  // green flash
  } catch (e) {
    showFeedback(`Save failed: ${e.message}`, 'error');  // red flash
  }
}

function showFeedback(message, type) {
  const el = document.getElementById('save-feedback') ?? (() => {
    const div = document.createElement('div');
    div.id = 'save-feedback';
    div.style.cssText = 'position:fixed;top:10px;right:10px;padding:8px 16px;border-radius:4px;font-weight:bold;z-index:9999;';
    document.body.appendChild(div);
    return div;
  })();
  el.textContent = message;
  el.style.background = type === 'success' ? '#4caf50' : '#f44336';
  el.style.color = '#fff';
  el.style.display = 'block';
  setTimeout(() => el.style.display = 'none', 2500);
}
```

- [ ] Manual check: save an asset, green "Saved!" appears. Break the server, save again, red error appears.

- [ ] Commit:
```bash

git commit -m "feat: asset editor save feedback (success/error)"
```

### Task 6: Map maker — add undo/redo

- [ ] Check if undo/redo is already present:
```bash
grep -n "undo\|redo\|history\|Ctrl.*Z\|keydown" auxserver/static/mapmaker.js | head -20
```

- [ ] If missing, add a simple command history:
```javascript
const history = [];
let historyIndex = -1;

function recordAction(action) {
  // action = { do: fn, undo: fn }
  history.splice(historyIndex + 1);  // discard redo stack
  history.push(action);
  historyIndex = history.length - 1;
  action.do();
}

document.addEventListener('keydown', e => {
  if ((e.ctrlKey || e.metaKey) && e.key === 'z') {
    if (historyIndex >= 0) { history[historyIndex].undo(); historyIndex--; }
  }
  if ((e.ctrlKey || e.metaKey) && e.key === 'y') {
    if (historyIndex < history.length - 1) { historyIndex++; history[historyIndex].do(); }
  }
});
```
  Wrap tile paint operations in `recordAction({ do: () => paintTile(...), undo: () => paintTile(prev) })`.

- [ ] Manual check: paint tiles, Ctrl+Z undoes, Ctrl+Y redoes.

- [ ] Commit:
```bash

git commit -m "feat: map maker undo/redo (Ctrl+Z / Ctrl+Y)"
```
